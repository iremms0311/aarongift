  <script>
    // --- EKRAN GEÇİŞ YÖNETİMİ ---
    function showScreen(screenId) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      document.getElementById(screenId).classList.add('active');
    }

    // --- 1. ŞİFRE SİSTEMİ ---
    let enteredCode = "";
    const correctCode = "0807";

    function pressKey(num) {
      if (enteredCode.length < 4) {
        enteredCode += num;
        updateDots();
      }
    }

    function clearKey() {
      enteredCode = "";
      updateDots();
    }

    function updateDots() {
      for (let i = 0; i < 4; i++) {
        const dot = document.getElementById(`dot-${i}`);
        if (i < enteredCode.length) {
          dot.classList.add('filled');
        } else {
          dot.classList.remove('filled');
        }
      }
    }

    function checkCode() {
      if (enteredCode === correctCode) {
        showScreen('screen-2');
        clearKey();
      } else {
        // Hatalı girmede kilit efekti / sıfırlama
        clearKey();
      }
    }

    // --- 5. ŞARKI SİSTEMİ ---
    const music = document.getElementById('bg-music');
    const vinyl = document.getElementById('vinyl');

    function toggleMusic() {
      if (music.paused) {
        music.play();
        vinyl.classList.add('playing');
      } else {
        music.pause();
        vinyl.classList.remove('playing');
      }
    }

    function pauseMusic() {
      music.pause();
      vinyl.classList.remove('playing');
    }

    // --- 6. CATCH MY HEART OYUN SİSTEMİ ---
    let score = 0;
    let moveInterval = null;
    const target = document.getElementById('heart-target');
    const hud = document.getElementById('hud');
    const gameOverMsg = document.getElementById('game-over');
    const gameArea = document.getElementById('game-area');

    function startGame() {
      resetGame();
      moveHeartRandomly();
      // Kalp 450ms'de bir hızlıca yer değiştirir
      moveInterval = setInterval(moveHeartRandomly, 450);
    }

    function moveHeartRandomly() {
      if (score >= 5) return;
      
      const maxX = gameArea.clientWidth - 40;
      const maxY = gameArea.clientHeight - 40;
      
      const randomX = Math.floor(Math.random() * maxX);
      const randomY = Math.floor(Math.random() * maxY);

      target.style.left = randomX + 'px';
      target.style.top = randomY + 'px';
    }

    function catchHeart() {
      if (score < 5) {
        score++;
        hud.innerText = `Caught: ${score}/5`;
        moveHeartRandomly();

        if (score === 5) {
          clearInterval(moveInterval);
          target.style.display = 'none';
          hud.style.display = 'none';
          gameOverMsg.style.display = 'block';
        }
      }
    }

    function resetGame() {
      score = 0;
      clearInterval(moveInterval);
      hud.innerText = `Caught: 0/5`;
      hud.style.display = 'block';
      target.style.display = 'block';
      gameOverMsg.style.display = 'none';
    }
  </script>
